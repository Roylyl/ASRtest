cmake_minimum_required(VERSION 3.20)
project(ASRNano LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_HTTPLIB OFF CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../Vendor/Fun-ASR/third_party/llama.cpp" llama)
add_library(CNano SHARED Sources/CNano/nano_core.cpp)
target_include_directories(CNano PRIVATE Sources/CNano/include)
target_link_libraries(CNano PRIVATE llama ggml)
set_target_properties(CNano PROPERTIES
    FRAMEWORK TRUE
    FRAMEWORK_VERSION A
    MACOSX_FRAMEWORK_IDENTIFIER org.localasr.nano
    VERSION 1.0.0
    SOVERSION 1
    PUBLIC_HEADER "${CMAKE_CURRENT_LIST_DIR}/Sources/CNano/include/asr_nano.h"
    XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED NO
    BUILD_WITH_INSTALL_RPATH TRUE
    INSTALL_NAME_DIR "@rpath"
    INSTALL_RPATH "@loader_path/Frameworks")
if(APPLE)
    target_link_options(CNano PRIVATE "LINKER:-exported_symbols_list,${CMAKE_CURRENT_LIST_DIR}/nano-exports.txt")
endif()
